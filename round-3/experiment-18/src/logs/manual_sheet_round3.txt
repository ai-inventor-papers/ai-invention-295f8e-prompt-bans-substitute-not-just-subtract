    from collections import defaultdict

    # Build adjacency list
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))

    # Memoization cache: key = node, value = shortest distance from s to this node
    min_distances = {node: float('inf') for node in range(n)}

    reached_t = False
    shortest_distance = float('inf')

    while stack:
        node, dist = stack.pop()

        if node == t:
            shortest_distance = min(shortest_distance, dist)
            reached_t = True

        for neighbor, weight in graph[node]:

                min_distances[neighbor] = new_dist
                stack.append((neighbor, new_dist))
