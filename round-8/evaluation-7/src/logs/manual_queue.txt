def solve(nums, k):
    # Count occurrences of each number
    count = {}
    for num in nums:
        count[num] = count.get(num, 0) + 1

    # Sort the unique numbers
    sorted_nums = sorted(count.keys())

    # Find the k-th smallest value
    for num in sorted_nums:
        if k <= count[num]:
            return num
        k -= count[num]

    # Use a deque for 0-1 BFS
    dq = deque()
    dq.append(s)
    
    # Create adjacency list
    adj = [[] for _ in range(n)]
    for u, v, w in edges:
        adj[u].append((v, w))
    
    while dq:
        u = dq.popleft()
        for v, w in adj[u]:
            if dist[v] > dist[u] + w:
                dist[v] = dist[u] + w
                if w == 0:
                    dq.appendleft(v)
                else:
                    dq.append(v)
